@override
Widget build(BuildContext context) {
  final items = menuItems.toList();

  return MenuAnchor(
    useRootOverlay: true,
    style: MenuStyle(
      // Prefer below; MenuAnchor flips above near the bottom edge.
      alignment: AlignmentDirectional.bottomStart,
      elevation: const WidgetStatePropertyAll(8),
      maximumSize: WidgetStatePropertyAll(
        Size(maxWidth ?? 224, double.infinity),
      ),
    ),
    menuChildren: [
      for (var index = 0; index < items.length; index++) ...[
        MenuItemButton(
          onPressed: items[index].$2
              ? () => menuSelected(items[index].$1)
              : null,
          style: const ButtonStyle(
            padding: WidgetStatePropertyAll(EdgeInsets.zero),
            alignment: AlignmentDirectional.centerStart,
          ),
          child: builder(items[index].$1),
        ),
        if (index != items.length - 1)
          const Divider(height: 1),
      ],
    ],
    builder: (
      BuildContext context,
      MenuController controller,
      Widget? child,
    ) {
      return IconButton(
        tooltip: tooltip,
        style: style ?? EvTheme.get.iconSecondary,
        onPressed: enableMenuButton
            ? () {
                controller.isOpen
                    ? controller.close()
                    : controller.open();
              }
            : null,
        icon: buttonWidget ??
            Icon(buttonIcon ?? Icons.more_vert),
      );
    },
  );
}
